## 패키지 분류
```
src/
├── datastructure/
│   ├── array_list/          // 배열, 리스트 기본 문제
│   ├── string/              // 문자열 처리
│   ├── stack/               // 스택
│   ├── queue/               // 큐
│   ├── deque/               // 덱
│   ├── hash/                // HashMap, HashSet
│   ├── heap/                // 힙, 우선순위 큐
│   ├── tree/                // 이진트리, BST, 트리 일반
│   ├── graph/               // 그래프 표현 및 기본 문제
│   ├── union_find/          // Union-Find, Disjoint Set
│   ├── trie/                // 트라이
│   └── segment_tree/        // 세그먼트 트리
│
└── algorithm/
    ├── sorting/             // 정렬 알고리즘
    ├── binary_search/       // 이분탐색
    ├── two_pointer/         // 투 포인터
    ├── sliding_window/      // 슬라이딩 윈도우
    ├── dfs/                 // 깊이 우선 탐색
    ├── bfs/                 // 너비 우선 탐색
    ├── shortest_path/       // 최단경로 (다익스트라, 벨만포드, 플로이드)
    ├── mst/                 // 최소신장트리 (크루스칼, 프림)
    ├── topological_sort/    // 위상정렬
    ├── greedy/              // 그리디
    ├── dp/                  // 동적계획법
    ├── backtracking/        // 백트래킹
    ├── divide_conquer/      // 분할정복
    ├── bruteforce/          // 완전탐색
    ├── simulation/          // 구현, 시뮬레이션
    ├── math/                // 수학, 정수론, 조합론
    ├── bit_manipulation/    // 비트마스킹
    ├── prefix_sum/          // 누적합, 구간합
    └── etc/                 // 기타
```
